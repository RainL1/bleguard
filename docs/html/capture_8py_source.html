19class CaptureSession:
20 """@brief Сессия живого захвата Bluetooth-трафика.
21
22 @details Управляет дочерним процессом tshark. Если tshark недоступен,
23 сессия переходит в ручной режим — файл не создаётся, но
24 считается запущенной.
25
26 @note Не потокобезопасен; вызывайте start() и stop() из одного потока.
27 """

29 def __init__(self):
30 """@brief Инициализирует сессию в состоянии «остановлена»."""
31 self._proc: Optional[subprocess.Popen] = None

45 def start(self) -> bool:
46 """@brief Запускает захват трафика.
47
48 @details Создаёт временный файл и запускает tshark на интерфейсе
49 bluetooth-monitor. Если tshark не найден — ручной режим.
50
51 @return True при успешном запуске.
52 """
53 if self._running:
54 return True
55
56 tmpfile = tempfile.NamedTemporaryFile(suffix='.btsnoop', delete=False)
57 self._capture_file = tmpfile.name
58 tmpfile.close()
59
60 tshark_path = _find_tshark()
61 if tshark_path:
62 try:
63 cmd = [
64 tshark_path,
65 '-i', 'bluetooth-monitor',
66 '-F', 'pcap',
67 '-w', self._capture_file,
68 ]
69 self._proc = subprocess.Popen(
70 cmd,
71 stdin=subprocess.DEVNULL,
72 stdout=subprocess.DEVNULL,
73 stderr=subprocess.DEVNULL,
74 start_new_session=True,
75 )
76 self._running = True
77 self._emit('tshark capture started')
78 return True
79 except Exception as e:
80 self._emit(f'tshark error: {e}')
81
82 self._running = True
83 self._emit('Capture session started (manual mode)')
84 return True

86 def stop(self) -> Optional[str]:
87 """@brief Останавливает захват и возвращает путь к файлу.
88
89 @details Посылает SIGINT процессу tshark и ждёт до 3 секунд,
90 затем завершает принудительно.
91
92 @return Путь к файлу захвата или None.
93 """
94 self._running = False
95 if self._proc and self._proc.poll() is None:
96 try:
97 self._proc.send_signal(signal.SIGINT)
98 self._proc.wait(timeout=3)
99 except Exception:
100 try:
101 self._proc.kill()
102 except Exception:
103 pass
104 self._proc = None
105 return self._capture_file

127def _find_tshark() -> Optional[str]:
128 """@brief Ищет исполняемый файл tshark в стандартных местах.
129
130 @return Путь к tshark или None, если не найден.
131 """
132 for path in ['/usr/bin/tshark', '/usr/local/bin/tshark', '/opt/homebrew/bin/tshark']:
133 if os.path.isfile(path):
134 return path
135 try:
136 result = subprocess.run(['which', 'tshark'], capture_output=True, text=True)
137 if result.returncode == 0:
138 return result.stdout.strip()
139 except Exception:
140 pass
141 return None
