62def _count_packets(path: str) -> int:
63 try:
64 with open(path, 'rb') as f:
65 data = f.read()
66 if data.startswith(b'btsnoop\x00'):
67 i, count = 16, 0
68 while i + 24 <= len(data):
69 incl_len = struct.unpack('>I', data[i + 4:i + 8])[0]
70 i += 24 + incl_len
71 count += 1
72 return count
73 elif data[:4] in (b'\xd4\xc3\xb2\xa1', b'\xa1\xb2\xc3\xd4'):
74 endian = '<' if data[:4] == b'\xd4\xc3\xb2\xa1' else '>'
75 i, count = 24, 0
76 while i + 16 <= len(data):
77 incl_len = struct.unpack(endian + 'I', data[i + 8:i + 12])[0]
78 i += 16 + incl_len
79 count += 1
80 return count
81 except Exception:
82 pass
83 return 0
