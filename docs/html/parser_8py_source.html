163def _parse_address(raw: bytes) -> str:
164 """@brief Преобразует 6-байтовый MAC в строку вида 'AA:BB:CC:DD:EE:FF'.
165
166 @param raw 6 байт в порядке контроллера (little-endian).

169 return ':'.join(f'{b:02X}' for b in reversed(raw))

172def _parse_btsnoop(data: bytes) -> list:
173 """@brief Разбирает BTSnoop-файл и возвращает список кортежей (flags, payload).
174
175 @param data Содержимое файла в байтах.
176 @return Список кортежей (flags: int, packet: bytes).
177 @throws ValueError Если файл не является BTSnoop или версия не поддерживается.
178 """
179 if not data.startswith(BTSNOOP_MAGIC):
180 raise ValueError('Not a valid BTSnoop file')
181
182 version = struct.unpack('>I', data[8:12])[0]
183 if version != BTSNOOP_VERSION:
184 raise ValueError(f'Unsupported BTSnoop version: {version}')
185
186 records = []
187 offset = 16
188 while offset + 24 <= len(data):
189 orig_len, incl_len, flags, drops = struct.unpack('>IIII', data[offset:offset+16])
190 ts_high, ts_low = struct.unpack('>II', data[offset+16:offset+24])
191 if offset + 24 + incl_len > len(data):
192 break
193 pkt = data[offset+24:offset+24+incl_len]
194 records.append((flags, pkt))
195 offset += 24 + incl_len
196
197 return records

200def _parse_btmon_packet(pkt: bytes) -> Optional[bytes]:
201 """@brief Извлекает HCI-полезную нагрузку из пакета PCAP с linktype 254 (btmon).
202
203 @param pkt Сырые байты пакета.
204 @return Байты HCI-полезной нагрузки или None, если opcode не поддерживается.
205 """
206 if len(pkt) < 4:
207 return None
208 opcode = struct.unpack('>H', pkt[2:4])[0]
209 payload = pkt[4:]
210 if opcode in (0x0002, 0x0003):
211 return payload
212 elif opcode in (0x0004, 0x0005):
213 return bytes([0x02]) + payload
214 return None

217def _parse_pcap(data: bytes) -> list:
218 """@brief Разбирает PCAP-файл и возвращает список кортежей (flags, payload).
219
220 @param data Содержимое файла в байтах.
221 @return Список кортежей (flags: int, packet: bytes).
222 @throws ValueError Если файл не является корректным PCAP.
223 """
224 magic = struct.unpack('<I', data[:4])[0]
225 if magic == 0xA1B2C3D4:
226 endian = '<'
227 elif magic == 0xD4C3B2A1:
228 endian = '>'
229 else:
230 raise ValueError('Not a valid pcap file')
231
232 network = struct.unpack(endian + 'I', data[20:24])[0]
233 is_btmon = (network == 254)
234
235 records = []
236 offset = 24
237 while offset + 16 <= len(data):
238 ts_sec, ts_usec, incl_len, orig_len = struct.unpack(endian + 'IIII', data[offset:offset+16])
239 if offset + 16 + incl_len > len(data):
240 break
241 pkt = data[offset+16:offset+16+incl_len]
242 if is_btmon:
243 payload = _parse_btmon_packet(pkt)
244 if payload is not None:
245 records.append((0x02, payload))
246 else:
247 records.append((0x02, pkt))
248 offset += 16 + incl_len
249
250 return records

253def _decode_smp(cid: int, smp_data: bytes) -> Optional[SmpPairing]:
254 """@brief Декодирует SMP Pairing Request/Response из L2CAP-данных.

276def _try_parse_l2cap(payload: bytes):
277 """@brief Ищет L2CAP-фрейм внутри сырой HCI-полезной нагрузки.
278
279 @details Перебирает до 6 смещений байт, проверяя длину L2CAP и CID.
280
281 @param payload Сырые байты HCI-пакета.
282 @return Кортеж (cid, smp_data) при успехе или (None, None).
283 """
284 for start in range(min(6, len(payload) - 4)):
285 if start + 4 > len(payload):
286 break
287 try:
288 l2cap_len = struct.unpack('<H', payload[start:start+2])[0]
289 cid = struct.unpack('<H', payload[start+2:start+4])[0]
290 if cid in (0x0004, 0x0006, 0x0005) and l2cap_len < 300:
291 smp_data = payload[start+4:start+4+l2cap_len]
292 return cid, smp_data
293 except Exception:
294 pass
295 return None, None

298def parse_capture(filepath: str) -> ParseResult:
299 """@brief Разбирает файл захвата BTSnoop или PCAP.
300
301 @details Определяет формат по магическим байтам, затем обходит все HCI-записи.
302 Ошибки добавляются в ParseResult.errors без исключений.
303
304 @param filepath Путь к файлу захвата.
305 @return Заполненный ParseResult.
306 """
307 result = ParseResult()
308
309 with open(filepath, 'rb') as f:
310 data = f.read()
311
312 try:
313 if data.startswith(BTSNOOP_MAGIC):
314 records = _parse_btsnoop(data)
315 elif data[:4] in (b'\xd4\xc3\xb2\xa1', b'\xa1\xb2\xc3\xd4'):
316 records = _parse_pcap(data)
317 else:
318 result.errors.append('Unknown file format')
319 return result
320 except Exception as e:
321 result.errors.append(str(e))
322 return result
323
324 result.raw_records_count = len(records)
325
326 for flags, pkt in records:
327 if len(pkt) < 3:
328 continue
329
330 _process_record(pkt, flags, result)
331
332 if result.connection is not None:
333 result.is_ble = True
334 result.connection_type = 'BLE'
335 elif result.encryption.ltk is not None:
336 result.is_ble = True
337 result.connection_type = 'BLE'
338
339 if result.is_classic_bt and not result.is_ble:
340 result.connection_type = 'Classic Bluetooth (BR/EDR)'
341
342 _finalize_key_size(result)
343
344 return result

347def _process_record(pkt: bytes, flags: int, result: ParseResult):
348 """@brief Направляет HCI-запись в нужный обработчик по первому байту пакета.
349
350 @param pkt Байты HCI-пакета вместе с байтом-индикатором.
351 @param flags Флаги направления BTSnoop.
352 @param result Строящийся ParseResult.
353 """
354 if len(pkt) < 2:
355 return
356
357 first = pkt[0]
358
359 if first == 0x01 and len(pkt) >= 3:
360 opcode = struct.unpack('<H', pkt[0:2])[0]
361 _handle_hci_cmd(opcode, pkt[2:], result)
362 return
363
364 if first == 0x04 and len(pkt) >= 2:
365 evt_code = pkt[1]
366 _handle_hci_evt(evt_code, pkt[3:] if len(pkt) > 3 else b'', result)
367 return
368
369 if first == 0x02 and len(pkt) >= 5:
370 _handle_hci_acl(pkt[1:], result)
371 return
372
373 if first == 0x19 and len(pkt) >= 3:
374 opcode = struct.unpack('<H', pkt[0:2])[0]
375 if opcode == 0x2019:
376 _handle_le_enable_encryption(pkt[3:], result)
377 return
378
379 if first == 0x3E and len(pkt) >= 3:
380 _handle_hci_evt(0x3E, pkt[2:], result)
381 return
382
383 if first == 0x08 and len(pkt) >= 5:
384 if pkt[1] == 0x04:
385 _handle_hci_evt(0x08, pkt[2:], result)
386 return
387
388 if first == 0x0F and len(pkt) >= 3:
389 if pkt[1] == 0x04:
390 opcode = struct.unpack('<H', pkt[3:5])[0] if len(pkt) >= 5 else 0
391 if opcode == 0x2019:
392 pass
393 return
394
395 cid, smp_data = _try_parse_l2cap(pkt)
396 if cid == 0x0006 and smp_data:
397 pairing = _decode_smp(cid, smp_data)
398 if pairing:
399 _apply_smp(pairing, result)
400 return
401
402 if len(pkt) >= 4:
403 for offset in range(min(6, len(pkt) - 8)):
404 if offset + 8 > len(pkt):
405 break
406 try:
407 maybe_handle = struct.unpack('<H', pkt[offset:offset+2])[0] & 0x0FFF
408 if maybe_handle == 0:
409 continue
410 maybe_cid = struct.unpack('<H', pkt[offset+4:offset+6])[0]
411 if maybe_cid == 0x0006:
412 smp_start = offset + 6
413 if smp_start + 1 < len(pkt):
414 smp_data_candidate = pkt[smp_start:]
415 p = _decode_smp(0x0006, smp_data_candidate)
416 if p:
417 _apply_smp(p, result)
418 break
419 except Exception:
420 pass

423def _handle_hci_cmd(opcode: int, params: bytes, result: ParseResult):
424 """@brief Обрабатывает HCI Command.
425
426 @details Распознаёт: 0x0405 BR/EDR Create Connection, 0x2019 LE Enable Encryption,
427 0x2006 LE Create Connection.
428
429 @param opcode 16-битный код команды HCI.
430 @param params Параметры команды.
431 @param result Строящийся ParseResult.
432 """
433 if opcode == 0x0405:
434 result.is_classic_bt = True
435 if not result.is_ble:
436 result.connection_type = 'Classic Bluetooth (BR/EDR)'
437 elif opcode == 0x2019 and len(params) >= 26:
438 _handle_le_enable_encryption(params, result)
439 elif opcode == 0x2006 and len(params) >= 13:
440 handle = struct.unpack('<H', params[1:3])[0]
441 addr_type = params[3]
442 addr_raw = params[4:10]
443 addr_str = _parse_address(addr_raw)
444 interval = struct.unpack('<H', params[10:12])[0] if len(params) >= 12 else 0
445 result.connection = LeConnectionInfo(
446 handle=handle,
447 peer_address=addr_str,
448 address_type=addr_type,
449 connection_interval=interval,
450 )
451 result.is_ble = True
452 result.connection_type = 'BLE'

474def _handle_hci_evt(evt_code: int, params: bytes, result: ParseResult):
475 """@brief Обрабатывает HCI Event.
476
477 @details Распознаёт: 0x03 BR/EDR Connection Complete, 0x3E LE Meta
478 (подсобытие 0x01 — LE Connection Complete), 0x08 Encryption Change.
479
480 @param evt_code Код события.
481 @param params Параметры события.
482 @param result Строящийся ParseResult.
483 """
484 if evt_code == 0x03 and len(params) >= 1:
485 result.is_classic_bt = True
486 if not result.is_ble:
487 result.connection_type = 'Classic Bluetooth (BR/EDR)'
488 return
489
490 if evt_code == 0x3E and len(params) >= 2:
491 subevent = params[0]
492 if subevent == 0x01 and len(params) >= 17:
493 status = params[1]
494 if status != 0x00:
495 return
496 handle = struct.unpack('<H', params[2:4])[0] if len(params) >= 4 else 0
497 role = params[4] if len(params) > 4 else 0
498 addr_type = params[5] if len(params) > 5 else 0
499 addr_raw = params[6:12] if len(params) >= 12 else bytes(6)
500 addr_str = _parse_address(addr_raw)
501 interval = struct.unpack('<H', params[12:14])[0] if len(params) >= 14 else 0
502 result.connection = LeConnectionInfo(
503 handle=handle,
504 peer_address=addr_str,
505 address_type=addr_type,
506 connection_interval=interval,
507 )
508 result.is_ble = True
509 result.connection_type = 'BLE'
510 elif evt_code == 0x08 and len(params) >= 4:
511 status = params[0]
512 handle = struct.unpack('<H', params[1:3])[0] if len(params) >= 3 else 0
513 encryption_enabled = params[3] if len(params) > 3 else 0
514 if status == 0x00:
515 result.encryption.enabled = bool(encryption_enabled)
