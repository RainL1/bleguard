226def set_lang(lang: str):
227 """@brief Устанавливает язык интерфейса.
228
229 @param lang Код языка: LANG_RU ('ru') или LANG_EN ('en').
230 """
231 global _current_lang
232 if lang in (LANG_RU, LANG_EN):
233 _current_lang = lang

237 """@brief Возвращает текущий язык интерфейса.

244def t(key: str) -> str:
245 """@brief Возвращает переведённую строку по ключу.
246
247 @details Порядок: текущий язык → английский → сам ключ как заглушка.
248
249 @param key Ключ из словаря STRINGS.
250 @return Локализованная строка.
251 """
252 entry = STRINGS.get(key, {})
253 return entry.get(_current_lang, entry.get(LANG_EN, key))
